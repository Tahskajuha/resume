% icon in blue circle; #1 -> icon
\newcommand*\icon[1]{\tikz[baseline=(char.base)]{
  \node[shape=circle,draw,inner sep=1pt, fill=mainblue,mainblue,text=white] (char) {#1};}
}

% Text in square node with rounded corners; #1 -> text, #2 -> color
\newcommand*\round[2]{%
\tikz[baseline=(char.base)]\node[anchor=north west, draw,rectangle,
  rounded corners, inner sep=1.6pt, minimum size=5.5mm, text height=3.6mm,
  fill=#2,#2,text=white](char){#1};%
}

\newcommand{\rendercvfield}[2]{%
	\ifthenelse{\equal{#1}{}}{}{#2}
}

\newcommand{\makeprofile}{%
	% grey bar on the left side
	\begin{tikzpicture}[remember picture,overlay]
		\node [rectangle, fill=asidecolor, anchor=north, minimum width=11cm, minimum height=\paperheight+1cm] (box) at (-5cm,0.5cm){};
	\end{tikzpicture}

	\begin{textblock}{7}(0.5, 0.2)
		\begin{flushleft}
			\hspace{13pt}
			
			% profile picture
			\begin{tikzpicture}
				\clip (\FormScale bp/2, \FormScale bp/2) circle[radius=\FormScale bp/2];
				\pdfrefxform\FormObj
			\end{tikzpicture}%
			
			% name
			{\Huge\color{mainblue}\cv@name}

			% jobtitle
			\begin{flushright}
				{\normalsize\color{black!80}\cv@jobtitle}
			\end{flushright}

			% table with icons 
			{
				\renewcommand{\arraystretch}{1.6}
				\begin{tabular}{c @{\hskip 0.2cm} p{5cm}}
					\rendercvfield{\cv@date}{\textsc{\Large\icon{\Info}} & \cv@date\\}
					\rendercvfield{\cv@address}{\textsc{\Large\icon{\Letter}} & \cv@address\\}
					\rendercvfield{\cv@numberphone}{\textsc{\Large\icon{\Telefon}} & \cv@numberphone\\}
					\rendercvfield{\cv@site}{\textsc{\Large\icon{\Mundus}} & \href{\cv@site}{\textcolor{cerulean}\cv@site}\\}
					\rendercvfield{\cv@mail}{\textsc{\large\icon{@}} & \href{mailto:\cv@mail}{\cv@mail}}
				\end{tabular}
			}

			\vspace{1em}

			% about me text
			\sidebarsection{About Me}{4.2cm}\\
			\cv@aboutme

			\vspace{1em}

			% skills
			\sidebarsection{Skill}{6cm}\\
			\cv@skills

		\end{flushleft}
	\end{textblock}
}

\newcounter{colorCounter}
\newcommand{\cvsectionlabel}[1]{
	{%
		% Switch between blue and gray
		\round{#1}{%
			\ifodd\value{colorCounter}%
				mainblue%
			\else%
				maingray%
			\fi%
		}%
	}%
	\stepcounter{colorCounter}%
}

\newcommand{\cvsection}[2]{
	\par%
	\hspace{0.5cm}
	{%
		\LARGE
		\color{headercolor}%
		\cvsectionlabel{#1}#2%
	}
	\par\vspace{\parskip}
}

	\hfill
	\begin{minipage}[t]{\dimexpr\textwidth-\timelinelabelwidth-1em\relax}
		{\color{headercolor}\bfseries #3}\par
		{\small\itshape #4}\par\vspace{0.3em}
		#5
	\end{minipage}
	\par\medskip
}
